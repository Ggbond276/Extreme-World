```csharp
public Result AcceptQuest(int questId, out Quest quest)
{
    quest = null;

    // 校验 1:防重(内存里已有)
    if (this.Quests.ContainsKey(questId))
        return Result.Failed;

    // 校验 2:配表存在性
    if (!DataManager.Instance.Quests.ContainsKey(questId))
        return Result.Failed;

    // EF6 创建一行(尚未提交)
    TCharacterQuest dbQuest = DBService.Instance.Entities.CharacterQuests.Create();
    dbQuest.TCharacterID = this.Owner.Data.ID;
    dbQuest.QuestID = questId;

    quest = new Quest(dbQuest);
    quest.InitQuetsDefine();                              // 根据 Target1 类型决定 Status

    this.Quests.Add(quest.QuestId, quest);                 // 内存字典更新
    this.Owner.Data.Quests.Add(dbQuest);                   // EF6 关联表更新

    DBService.Instance.save();                             // ★ 真写库
    return Result.Success;
}
```

### D3. SubmitQuest

**文件**:`Managers/QuestManager.cs:89-144`

```csharp
public Result SubmitQuest(int questId, out Quest quest)
{
    quest = null;
    if (!this.Quests.ContainsKey(questId)) return Result.Failed;
    if (!DataManager.Instance.Quests.ContainsKey(questId)) return Result.Failed;

    quest = this.Quests[questId];
    if (!quest.CanSubmit) return Result.Failed;            // 校验可交付

    // 发奖:直接修改 Owner 的运行时数据
    if (quest.Define.RewardGold > 0)
        this.Owner.Gold += quest.Define.RewardGold;         // ★ 走 Character.Gold setter
    if (quest.Define.RewardExp > 0)
        this.Owner.Exp += quest.Define.RewardExp;
    if (quest.Define.RewardItem1 > 0)
        this.Owner.ItemManager.AddItem(quest.Define.RewardItem1, quest.Define.RewardItem1Count);
    // ... RewardItem2/3 同理 ...

    quest.OnSubmit();                                      // DbQuest.Status = Finished
    DBService.Instance.save();
    return Result.Success;
}
```

```csharp
public void OnQuestAccept(NetConnection<NetSession> sender, QuestAcceptRequest request)
{
    Character character = sender.Session.Character;        // ★ 从 Session 拿角色
    int questId = request.QuestId;

    Result result = character.questManager.AcceptQuest(questId, out Quest quest);

    sender.Session.Response.questAccept = new QuestAcceptResponse();
    if (result == Result.Success && quest != null)
        sender.Session.Response.questAccept.Quest = quest.ToNQuestInfo();
    else
        sender.Session.Response.questAccept.Errormsg = "条件不足 任务无法接取";

    sender.Session.Response.questAccept.Result = result;
    sender.SendResponse();                                  // ★ 发包
}
```

### F2. Init + InitQuests

```csharp
public void Init(List<NQuestInfo> quests)
{
    this.quests = quests;
    allQuests.Clear();
    npcQuests.Clear();
    this.InitQuests();
}

public void InitQuests()
{
    // 第一轮:把服务器下发的任务(已接)装进字典
    foreach(var Info in quests)
    {
        Quest quest = new Quest(Info);
        this.allQuests[quest.Define.ID] = quest;
        this.AddNpcQuest(quest.Define.AcceptNPC, quest);
        this.AddNpcQuest(quest.Define.SubmitNPC, quest);
    }

    // 第二轮:遍历所有 QuestDefine,过滤出"可接"任务
    foreach(var kv in DataManager.Instance.Quests)
    {
        QuestDefine def = kv.Value;
        // 职业过滤
        if (def.LimitClass != CharacterClass.None && def.LimitClass != User.Instance.CurrentCharacter.Class) continue;
        // 等级过滤
        if (def.LimitLevel > User.Instance.CurrentCharacter.Level) continue;
        // 已接过滤
        if (allQuests.ContainsKey(def.ID)) continue;
        // 前置任务过滤
        if (def.PreQuest > 0)
        {
            if (!allQuests.TryGetValue(def.PreQuest, out Quest preQuest)) continue;
            if (preQuest.Info == null) continue;
            if (preQuest.Info.Status != QuestStatus.Finished) continue;
        }

        Quest quest = new Quest(def);
        allQuests.Add(def.ID, quest);
        this.AddNpcQuest(def.AcceptNPC, quest);
        this.AddNpcQuest(def.SubmitNPC, quest);
    }
}
```

### F3. OpenNpcQuest(优先级决策)

```csharp
public bool OpenNpcQuest(int npcId)
{
    if(this.npcQuests.TryGetValue(npcId, out List<Quest> questList))
    {
        Quest targetQuest = null;

        // 优先级 1:可交付
        foreach (var quest in questList)
            if (npcId == quest.Define.SubmitNPC && quest.Info != null &&
                quest.Info.Status == QuestStatus.Complated)
            { targetQuest = quest; break; }

        // 优先级 2:可接取
        if (targetQuest == null)
            foreach (var quest in questList)
                if (npcId == quest.Define.AcceptNPC && quest.Info == null)
                { targetQuest = quest; break; }

        // 优先级 3:进行中催更
        if (targetQuest == null)
            foreach (var quest in questList)
                if (npcId == quest.Define.SubmitNPC && quest.Info != null &&
                    quest.Info.Status == QuestStatus.InProgress)
                { targetQuest = quest; break; }

        if (targetQuest != null)
        {
            if (targetQuest.Info != null && targetQuest.Info.Status == QuestStatus.InProgress)
            {
                MessageBox.Show(targetQuest.Define.DialogIncomplete ?? "还有任务没有完成哦");
            }
            else
            {
                OnOpenQuestDialog?.Invoke(targetQuest);
            }
            return true;
        }
    }
    return false;
}
```

### F4. NPC 头顶图标状态

```csharp
public NpcQuestStatus GetQuestStatusByNpc(int npcId)
{
    if (this.npcQuests.TryGetValue(npcId, out List<Quest> questList))
    {
        // 找可交付(Complete)
        foreach(var quest in questList)
            if (quest.Define.SubmitNPC == npcId && quest.Info != null &&
                quest.Info.Status == QuestStatus.Complated)
                return NpcQuestStatus.Complete;

        // 找可接取(Available)
        foreach(var quest in questList)
            if (quest.Define.AcceptNPC == npcId && quest.Info == null)
                return NpcQuestStatus.Available;

        // 找进行中(Incomplete)
        foreach(var quest in questList)
            if (quest.Define.SubmitNPC == npcId && quest.Info != null &&
                quest.Info.Status == QuestStatus.InProgress)
                return NpcQuestStatus.Incomplete;
    }
    return NpcQuestStatus.None;
}
```
